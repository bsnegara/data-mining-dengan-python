\begin{tikzpicture}[dl,x=7mm,y=7mm]
  % kubus 3 x 3 x 3 dalam proyeksi oblik: kota (bawah), bulan (kanan), kategori (dalam)
  \def\dx{0.45}\def\dy{0.3}
  \foreach \k in {2,1,0}{
    \foreach \i in {0,1,2}{\foreach \j in {0,1,2}{
      \pgfmathsetmacro{\x}{\i+\k*\dx}\pgfmathsetmacro{\y}{\j+\k*\dy}
      \ifnum\k=0
        \draw[draw=dlBiru!70!black,fill=dlBiru!12] (\x,\y) rectangle ++(1,1);
      \fi}}}
  \foreach \i in {0,1,2}{ % atap
    \foreach \k in {0,1,2}{
      \pgfmathsetmacro{\x}{\i+\k*\dx}\pgfmathsetmacro{\y}{3+\k*\dy}
      \draw[draw=dlBiru!70!black,fill=dlBiru!25] (\x,\y) -- ++(\dx,\dy) -- ++(1,0) -- ++(-\dx,-\dy) -- cycle;}}
  \foreach \j in {0,1,2}{ % sisi kanan
    \foreach \k in {0,1,2}{
      \pgfmathsetmacro{\x}{3+\k*\dx}\pgfmathsetmacro{\y}{\j+\k*\dy}
      \draw[draw=dlBiru!70!black,fill=dlBiru!35] (\x,\y) -- ++(\dx,\dy) -- ++(0,1) -- ++(-\dx,-\dy) -- cycle;}}
  % sorotan irisan (slice): kolom bulan Januari
  \draw[dlOranye,line width=1.3pt] (0,0) rectangle (1,3);
  \node[keterangan,below] at (1.5,0) {kota};
  \node[keterangan,rotate=90,above] at (0,1.5) {bulan};
  \node[keterangan,rotate=34] at (4.05,3.6) {kategori};
  \node[keterangan,anchor=west,text=dlOranye] at (5.2,2.6) {\textbf{irisan}: bulan = Januari};
  \node[keterangan,anchor=west,align=left] at (5.2,1.6) {\textbf{dadu}: subkubus,\\mis.\ Ramadan $\times$ 2 kota $\times$ sore};
  \node[keterangan,anchor=west,align=left] at (5.2,0.35) {\textbf{roll-up}: gerai $\to$ kota $\to$ nasional\\\textbf{drill-down}: arah sebaliknya};
\end{tikzpicture}
